<script>
(() => {
"use strict";
const SIZE = 20;
const STEP_MS = 125;
const board = document.getElementById("board");
const scoreEl = document.getElementById("score");
const statusEl = document.getElementById("status");
const overlay = document.getElementById("overlay");
const overlayTitle = document.getElementById("overlayTitle");
const overlayText = document.getElementById("overlayText");
const startBtn = document.getElementById("startBtn");
const pauseBtn = document.getElementById("pauseBtn");
const restartBtn = document.getElementById("restartBtn");
const cells = [];
for (let i = 0; i < SIZE * SIZE; i++) {
const cell = document.createElement("div");
cell.className = "cell";
board.appendChild(cell);
cells.push(cell);
}
const directions = {
up: { x: 0, y: -1 },
down: { x: 0, y: 1 },
left: { x: -1, y: 0 },
right: { x: 1, y: 0 }
};
let snake, direction, queuedDirection, food, score;
let timer = null;
let state = "ready"; // ready, running, paused, over
function indexOf(point) {
return point.y * SIZE + point.x;
}
function randomFood() {
const empty = [];
for (let y = 0; y < SIZE; y++) {
for (let x = 0; x < SIZE; x++) {
if (!snake.some(part => part.x === x && part.y === y)) {
empty.push({ x, y });
}
}
}
return empty.length
? empty[Math.floor(Math.random() * empty.length)]
: null;
}
function render() {
for (const cell of cells) cell.className = "cell";
if (food) cells[indexOf(food)].classList.add("food");
snake.forEach((part, i) => {
cells[indexOf(part)].classList.add(i === 0 ? "head" : "snake");
});
scoreEl.textContent = String(score);
}
function showOverlay(title, text) {
overlayTitle.textContent = title;
overlayText.textContent = text;
overlay.hidden = false;
}
function hideOverlay() {
overlay.hidden = true;
}
function reset() {
clearInterval(timer);
timer = null;
snake = [
{ x: 9, y: 10 },
{ x: 8, y: 10 },
{ x: 7, y: 10 }
];
direction = directions.right;
queuedDirection = null;
score = 0;
food = randomFood();
state = "ready";
render();
statusEl.textContent = "시작 버튼을 눌러 게임을 시작하세요.";
startBtn.textContent = "게임 시작";
pauseBtn.textContent = "일시정지";
showOverlay("준비됐나요?", "시작 버튼을 눌러 게임을 시작하세요.");
}
function start() {
if (state === "running") return;
if (state === "over") reset();
state = "running";
hideOverlay();
statusEl.textContent = "게임 진행 중";
startBtn.textContent = "게임 중";
timer = setInterval(tick, STEP_MS);
}
function pause() {
if (state === "running") {
clearInterval(timer);
timer = null;
state = "paused";
statusEl.textContent = "일시정지";
startBtn.textContent = "계속하기";
pauseBtn.textContent = "계속하기";
showOverlay("일시정지", "시작 버튼 또는 계속하기를 눌러 이어서 플레이하세요.");
} else if (state === "paused") {
start();
pauseBtn.textContent = "일시정지";
}
}
function endGame(won = false) {
clearInterval(timer);
timer = null;
state = "over";
statusEl.textContent = won ? "게임판을 모두 채웠습니다!" : "게임 종료";
startBtn.textContent = "게임 시작";
pauseBtn.textContent = "일시정지";
showOverlay(won ? "완벽해요!" : "게임 오버", `최종 점수: ${score}점 · 다시 시작을 눌러 재도전하세요.`);
}
function tick() {
if (queuedDirection) {
direction = queuedDirection;
queuedDirection = null;
}
const head = snake[0];
const next = { x: head.x + direction.x, y: head.y + direction.y };
const eating = food && next.x === food.x && next.y === food.y;
if (next.x < 0 || next.x >= SIZE || next.y < 0 || next.y >= SIZE) {
endGame();
return;
}
// 먹이를 먹지 않는 이동에서는 꼬리가 한 칸 빠지므로, 빠질 꼬리 칸 진입은 허용합니다.
const bodyToCheck = eating ? snake : snake.slice(0, -1);
if (bodyToCheck.some(part => part.x === next.x && part.y === next.y)) {
endGame();
return;
}
snake.unshift(next);
if (eating) {
score += 10;
food = randomFood();
if (!food) {
render();
endGame(true);
return;
}
} else {
snake.pop();
}
render();
}
function requestDirection(name) {
if (state === "over") return;
const next = directions[name];
if (!next) return;
// 현재 이동 방향(또는 이미 예약된 방향)의 정반대 전환은 무시합니다.
const reference = queuedDirection || direction;
if (next.x === -reference.x && next.y === -reference.y) return;
// 한 틱에 한 번만 회전 예약: 버튼을 연타해도 방향이 꼬이지 않습니다.
queuedDirection = next;
if (state === "ready") start();
}
startBtn.addEventListener("click", () => {
if (state === "paused") {
start();
pauseBtn.textContent = "일시정지";
} else {
start();
}
});
pauseBtn.addEventListener("click", pause);
restartBtn.addEventListener("click", () => {
reset();
start();
});
document.addEventListener("keydown", event => {
const keyMap = {
ArrowUp: "up", w: "up", W: "up",
ArrowDown: "down", s: "down", S: "down",
ArrowLeft: "left", a: "left", A: "left",
ArrowRight: "right", d: "right", D: "right"
};
if (keyMap[event.key]) {
event.preventDefault();
requestDirection(keyMap[event.key]);
} else if (event.code === "Space") {
event.preventDefault();
if (state === "running" || state === "paused") pause();
else if (state === "ready") start();
}
});
document.querySelectorAll("[data-dir]").forEach(button => {
const choose = event => {
event.preventDefault();
requestDirection(button.dataset.dir);
};
button.addEventListener("pointerdown", choose);
button.addEventListener("click", event => {
// 키보드로 버튼을 누른 경우도 지원합니다. 포인터 입력은 pointerdown에서 처리합니다.
if (event.detail === 0) requestDirection(button.dataset.dir);
});
});
reset();
})();
</script>
